# Auth, sessions and cookies

Short-lived JWT access token (15 min) + opaque refresh token stored only as a hash in Postgres, rotated on every
refresh, limited per user, sent to browsers in an httpOnly cookie scoped to `/api/auth`. katax-cli generates
`requireAuth`, `requireRole`, `jwt.utils` and `password.utils`, but not the auth routes: build them with this file.

## 1. Mounting auth

```ts
// api/routes.ts
router.get('/health', healthCheckHandler);
router.use('/auth', noStore, authRouter);   // public; responses carry tokens -> no-store
router.use(requireAuth);                     // everything below: req.user = { id, email, role }
router.use('/habits', habitsRouter);
router.use('/admin', requireRole('admin'), adminRouter);
```

One door. Per-router `authenticateToken` is acceptable only when one prefix mixes auth types (JWT for the
dashboard, API key for an SDK). Controllers never parse tokens: they receive `userId`.

## 2. Sessions table

```sql
CREATE TABLE user_sessions (
  id                 uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id            uuid NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  refresh_token_hash text NOT NULL UNIQUE,
  platform           text,               -- web | mobile | tauri
  user_agent         text,
  ip_address         text,
  expires_at         timestamptz NOT NULL,
  revoked_at         timestamptz,
  revoked_reason     text,               -- rotated | logout | logout_all | max_sessions_exceeded | password_changed
  created_at         timestamptz NOT NULL DEFAULT now(),
  last_used_at       timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX user_sessions_active_idx ON user_sessions (user_id) WHERE revoked_at IS NULL;
```

## 3. Repository

```ts
// api/auth/sessions.repository.ts
import { db } from '../../shared/db.js';
import type { SqlExecutor } from '../../shared/db.js';
import type { CountRow, IdRow, UserSessionRow } from '../../types/db.types.js';
import type { NewSessionParams } from './auth.types.js';

const SESSION_COLUMNS = 'id, user_id, platform, user_agent, ip_address, expires_at, revoked_at, created_at, last_used_at';

export async function insertSession(p: NewSessionParams, executor: SqlExecutor = db()): Promise<IdRow> {
  const rows = await executor.query<IdRow[]>(
    `INSERT INTO user_sessions (user_id, refresh_token_hash, platform, user_agent, ip_address, expires_at)
     VALUES ($1, $2, $3, $4, $5, now() + ($6 || ' milliseconds')::interval) RETURNING id`,
    [p.userId, p.refreshTokenHash, p.platform, p.userAgent, p.ipAddress, String(p.ttlMs)]
  );
  return rows[0]!;
}

/** Atomic claim: only one concurrent refresh with the same token gets the row (see concurrency.md). */
export async function claimSessionByHash(hash: string, executor: SqlExecutor = db()): Promise<UserSessionRow | null> {
  const rows = await executor.query<UserSessionRow[]>(
    `UPDATE user_sessions SET revoked_at = now(), revoked_reason = 'rotated'
      WHERE refresh_token_hash = $1 AND revoked_at IS NULL AND expires_at > now()
      RETURNING ${SESSION_COLUMNS}`,
    [hash]
  );
  return rows[0] ?? null;
}

export async function countActiveSessions(userId: string, executor: SqlExecutor = db()): Promise<number> {
  const rows = await executor.query<CountRow[]>(
    `SELECT COUNT(*)::text AS count FROM user_sessions WHERE user_id = $1 AND revoked_at IS NULL AND expires_at > now()`,
    [userId]
  );
  return Number(rows[0]?.count ?? 0);
}

export async function revokeOldestSessions(userId: string, howMany: number, executor: SqlExecutor = db()): Promise<void> {
  await executor.query(
    `UPDATE user_sessions SET revoked_at = now(), revoked_reason = 'max_sessions_exceeded'
      WHERE id IN (SELECT id FROM user_sessions WHERE user_id = $1 AND revoked_at IS NULL
                   ORDER BY last_used_at ASC LIMIT $2)
        AND revoked_at IS NULL`,
    [userId, howMany]
  );
}

export async function revokeSession(id: string, userId: string, reason: string, executor: SqlExecutor = db()): Promise<boolean> {
  const rows = await executor.query<IdRow[]>(
    `UPDATE user_sessions SET revoked_at = now(), revoked_reason = $3
      WHERE id = $1 AND user_id = $2 AND revoked_at IS NULL RETURNING id`,
    [id, userId, reason]
  );
  return rows.length > 0;
}
```

## 4. Login, refresh, logout

```ts
// api/auth/auth.controller.ts (excerpt)
const MAX_ACTIVE_SESSIONS = 5;
const REFRESH_TTL_MS = parseDurationToMs(process.env.JWT_REFRESH_EXPIRES_IN || '7d');

// generateRandomToken, sha256, verifyPassword: shared/password.utils.ts (generated by katax init)

/** Opaque refresh token: random, never a JWT; only its hash is stored. */
function newRefreshToken(): { token: string; hash: string } {
  const token = generateRandomToken(32);
  return { token, hash: sha256(token) };
}

async function issueSession(user: UserRow, ctx: AuthContext, tx: SqlExecutor) {
  await lockKey(tx, `sessions:${user.id}`);                     // concurrent logins can't exceed the limit
  const active = await sessionsRepository.countActiveSessions(user.id, tx);
  if (active >= MAX_ACTIVE_SESSIONS) {
    await sessionsRepository.revokeOldestSessions(user.id, active - MAX_ACTIVE_SESSIONS + 1, tx);
  }
  const refresh = newRefreshToken();
  await sessionsRepository.insertSession(
    { userId: user.id, refreshTokenHash: refresh.hash, platform: ctx.platform, userAgent: ctx.userAgent,
      ipAddress: ctx.ip, ttlMs: REFRESH_TTL_MS },
    tx
  );
  const accessToken = generateAccessToken({ userId: user.id, email: user.email, role: user.role });
  return { accessToken, refreshToken: refresh.token };
}

export async function login(input: LoginInput, ctx: AuthContext): Promise<ControllerResult<AccessTokenResult>> {
  try {
    const user = await usersRepository.findByEmail(input.email);
    // Same message and similar timing whether the email exists or not.
    const ok = await verifyPassword(input.password, user?.password_hash ?? DUMMY_HASH); // DUMMY_HASH: a fixed bcrypt hash
    if (!user || !ok || !user.is_active) throw unauthorizedError('Invalid email or password');
    const tokens = await withTransaction((tx) => issueSession(user, ctx, tx));
    return tokenResponse('Logged in', tokens, ctx.platform);
  } catch (error) {
    return toSafeErrorResult(error, { context: 'auth.login' });
  }
}

export async function refresh(refreshToken: string | undefined, ctx: AuthContext) {
  try {
    if (!refreshToken) throw unauthorizedError('Missing refresh token');
    const tokens = await withTransaction(async (tx) => {
      const old = await sessionsRepository.claimSessionByHash(sha256(refreshToken), tx);
      if (!old) throw unauthorizedError('Invalid or expired refresh token');
      const user = await usersRepository.findById(old.user_id, tx);
      if (!user?.is_active) throw unauthorizedError('Invalid or expired refresh token');
      return issueSession(user, ctx, tx);
    });
    return tokenResponse('Token refreshed', tokens, ctx.platform);
  } catch (error) {
    return toSafeErrorResult(error, { context: 'auth.refresh' });
  }
}
```

Logout revokes the presented session (`revokeSession(..., 'logout')`) and clears the cookie with
`getClearedRefreshCookieOptions()`. Changing the password revokes every session of the user except, optionally,
the current one.

## 5. Cookies and native clients

`assets/templates/config/cookie.config.ts`: `httpOnly`, `secure` in production, `path: '/api/auth'`, `sameSite`
from `COOKIE_SAME_SITE` (`lax` when the front end and the API share a registrable domain such as
`app.example.com` + `api.example.com`; `none` + `secure` only across different domains).

Controllers return cookies as data and `sendResponse` writes them, so set and clear always use the same options:

```ts
function tokenResponse(message: string, tokens: Tokens, platform: string | undefined) {
  const native = platform === 'mobile' || platform === 'tauri';
  return {
    ...createResponse.success(message, {
      access_token: tokens.accessToken,
      ...(native ? { refresh_token: tokens.refreshToken } : {})
    }),
    // ControllerResult.cookies is applied by sendResponse(); logout sends the same cookie with
    // getClearedRefreshCookieOptions() and an empty value.
    cookies: [{ name: REFRESH_COOKIE_NAME, value: tokens.refreshToken, options: getRefreshCookieOptions() }]
  };
}
```

Native clients (React Native, Tauri) can't use the httpOnly cookie: they send `x-platform: mobile|tauri` (add it
to CORS `allowedHeaders`) and only then receive the refresh token in the body. Web clients never do.

## 6. Session management endpoints

`GET /api/auth/sessions` (the user's active sessions, current one marked), `DELETE /api/auth/sessions/:id`,
`POST /api/auth/sessions/revoke-others`. Every query is scoped by `user_id`; never return hashes.

## 7. Optional

- **Passkeys (WebAuthn)** as a second login method: store credential id, public key, counter and transports per
  user; the registration and authentication challenges are one-time values (claim them atomically).
- **Email verification guard**: a middleware after `requireAuth` on routes that need a verified email, answering
  403 with a dedicated error code the front end can act on.

## 8. Prohibited

- `express-session` / `connect-redis`: sessions are rows in Postgres.
- Refresh tokens stored in clear text, or JWT refresh tokens with no server-side revocation.
- Revoking with `UPDATE ... WHERE id = $1` during rotation (two refreshes both succeed).
- `sameSite: 'none'` without `secure`, or a refresh cookie with `path: '/'`.
